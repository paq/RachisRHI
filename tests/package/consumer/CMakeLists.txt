cmake_minimum_required(VERSION 4.0)
option(RACHIS_PACKAGE_C_ONLY "Verify that the C ABI needs no C++ toolchain" OFF)
if(RACHIS_PACKAGE_C_ONLY)
    project(RachisPackageConsumer LANGUAGES C)
else()
    project(RachisPackageConsumer LANGUAGES C CXX)
endif()

find_package(Rachis 0.3 CONFIG REQUIRED)
enable_testing()

foreach(backend IN ITEMS D3D12 VULKAN METAL)
    get_target_property(rachis_enable_${backend} Rachis::RHI RACHIS_ENABLE_${backend})
    if(rachis_enable_${backend} MATCHES "-NOTFOUND$")
        message(FATAL_ERROR "Rachis::RHI does not export RACHIS_ENABLE_${backend}.")
    endif()
endforeach()
if(rachis_enable_D3D12 AND NOT TARGET Rachis::RHID3D12)
    message(FATAL_ERROR "The package enables D3D12 but does not export Rachis::RHID3D12.")
endif()
if(rachis_enable_VULKAN AND NOT TARGET Rachis::RHIVulkan)
    message(FATAL_ERROR "The package enables Vulkan but does not export Rachis::RHIVulkan.")
endif()
if(rachis_enable_METAL AND NOT TARGET Rachis::RHIMetal)
    message(FATAL_ERROR "The package enables Metal but does not export Rachis::RHIMetal.")
endif()
if(TARGET Rachis::volk AND CMAKE_DL_LIBS)
    get_target_property(rachis_volk_link_libraries Rachis::volk INTERFACE_LINK_LIBRARIES)
    if(NOT "${CMAKE_DL_LIBS}" IN_LIST rachis_volk_link_libraries)
        message(FATAL_ERROR "Rachis::volk does not link ${CMAKE_DL_LIBS}, which volk needs for dlopen.")
    endif()
endif()
if(rachis_enable_D3D12)
    set(rachis_package_backend 0)
    set(rachis_package_adapter 2)
elseif(rachis_enable_VULKAN)
    set(rachis_package_backend 1)
    set(rachis_package_adapter 0)
elseif(rachis_enable_METAL)
    set(rachis_package_backend 2)
    set(rachis_package_adapter 0)
else()
    message(FATAL_ERROR "The Rachis package enables no backend.")
endif()

add_executable(RachisPackageC main.c)
target_compile_features(RachisPackageC PRIVATE c_std_11)
target_link_libraries(RachisPackageC PRIVATE Rachis::C)
add_test(NAME RachisPackageC COMMAND RachisPackageC)

set(rachis_package_consumers RachisPackageC)
if(NOT RACHIS_PACKAGE_C_ONLY)
    add_executable(RachisPackageCpp main.cpp)
    target_compile_features(RachisPackageCpp PRIVATE cxx_std_23)
    target_link_libraries(RachisPackageCpp PRIVATE Rachis::RHI)
    add_test(NAME RachisPackageCpp COMMAND RachisPackageCpp)
    list(APPEND rachis_package_consumers RachisPackageCpp)
endif()

foreach(target IN LISTS rachis_package_consumers)
    target_compile_definitions(${target} PRIVATE
        RACHIS_PACKAGE_BACKEND=${rachis_package_backend}
        RACHIS_PACKAGE_ADAPTER=${rachis_package_adapter})
endforeach()

if(WIN32)
    set(rachis_package_runtime_dlls)
    foreach(target IN LISTS rachis_package_consumers)
        list(APPEND rachis_package_runtime_dlls "$<TARGET_RUNTIME_DLLS:${target}>")
    endforeach()
    add_custom_target(RachisPackageRuntimeDlls
        COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:RachisPackageC>"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<REMOVE_DUPLICATES:${rachis_package_runtime_dlls}>" "$<TARGET_FILE_DIR:RachisPackageC>"
        COMMAND_EXPAND_LISTS
        VERBATIM)
    add_dependencies(RachisPackageRuntimeDlls Rachis::C)
    foreach(target IN LISTS rachis_package_consumers)
        add_dependencies(${target} RachisPackageRuntimeDlls)
    endforeach()
endif()
